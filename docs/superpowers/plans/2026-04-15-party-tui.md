# PARTY TUI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a native Rust TUI application for managing AI agents with threaded conversations, persistent file-based storage, inline content blocks, and responsive layout.

**Architecture:** Single `App` state struct with unidirectional data flow. UI rendering split into focused sub-modules under `src/ui/`. Disk persistence via JSON files under `~/.party/agents/`. Async event loop via Tokio with Crossterm polling at 50ms.

**Tech Stack:** Rust 2021, Ratatui 0.29, Crossterm 0.28, Tokio 1, Serde/serde_json, Chrono 0.4, UUID v4, dirs 5, anyhow 1.

**Spec:** `docs/superpowers/specs/2026-04-15-party-tui-design.md` (in the ClaudeDataAgent repo)

```rust
use chrono::{DateTime, Utc};
use serde::{Deserialize, Serialize};
use uuid::Uuid;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Agent {
    pub id: Uuid,
    pub name: String,
    pub slug: String,
    pub role: String,
    pub status: AgentStatus,
    pub channels: Vec<String>,
    pub projects: Vec<Project>,
    pub context_snapshots: Vec<ContextSnapshot>,
    pub created_at: DateTime<Utc>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub enum AgentStatus {
    Active,
    Idle,
    Error,
    Offline,
}

impl AgentStatus {
    pub fn label(&self) -> &str {
        match self {
            AgentStatus::Active => "ACTIVE",
            AgentStatus::Idle => "IDLE",
            AgentStatus::Error => "ERROR",
            AgentStatus::Offline => "OFFLINE",
        }
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Thread {
    pub id: Uuid,
    pub label: String,
    pub messages: Vec<Message>,
    pub forked_from: Option<(Uuid, usize)>,
    pub created_at: DateTime<Utc>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Message {
    pub id: Uuid,
    pub role: Role,
    pub content: String,
    pub timestamp: DateTime<Utc>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub enum Role {
    You,
    Agent,
    System,
}

impl Role {
    pub fn label(&self) -> &str {
        match self {
            Role::You => "you",
            Role::Agent => "agent",
            Role::System => "sys",
        }
    }
}

#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
pub enum InlineBlock {
    Metric {
        label: String,
        value: String,
        delta: Option<String>,
    },
    Table {
        headers: Vec<String>,
        rows: Vec<Vec<String>>,
    },
    Approval {
        action: String,
        status: ApprovalStatus,
    },
    Code {
        language: Option<String>,
        code: String,
    },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub enum ApprovalStatus {
    Pending,
    Approved,
    Rejected,
}

impl ApprovalStatus {
    pub fn symbol(&self) -> &str {
        match self {
            ApprovalStatus::Pending => "?",
            ApprovalStatus::Approved => "✓",
            ApprovalStatus::Rejected => "✗",
        }
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Project {
    pub name: String,
    pub status: ProjectStatus,
    pub progress: f32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
pub enum ProjectStatus {
    Active,
    Paused,
    Done,
}

impl ProjectStatus {
    pub fn label(&self) -> &str {
        match self {
            ProjectStatus::Active => "ACTIVE",
            ProjectStatus::Paused => "PAUSED",
            ProjectStatus::Done => "DONE",
        }
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ContextSnapshot {
    pub label: String,
    pub saved_at: DateTime<Utc>,
    pub token_count: u64,
}

#[derive(Debug, Clone)]
pub struct FileNode {
    pub name: String,
    pub is_dir: bool,
    pub size: Option<u64>,
    pub children: Vec<FileNode>,
    pub expanded: bool,
}

/// Convert a display name to a URL-safe slug.
/// "Google Ads" → "google-ads"
pub fn slugify(name: &str) -> String {
    name.to_lowercase()
        .chars()
        .map(|c| if c.is_alphanumeric() { c } else if c == ' ' || c == '-' { '-' } else { '\0' })
        .filter(|c| *c != '\0')
        .collect::<String>()
        .split('-')
        .filter(|s| !s.is_empty())
        .collect::<Vec<&str>>()
        .join("-")
}
```

```rust
use crate::models::*;
use anyhow::{Context, Result};
use chrono::Utc;
use std::fs;
use std::path::{Path, PathBuf};
use uuid::Uuid;

/// Get the default party directory (~/.party/)
pub fn default_party_dir() -> PathBuf {
    dirs::home_dir()
        .expect("Could not determine home directory")
        .join(".party")
}

/// Initialize the ~/.party/ directory structure if it doesn't exist.
pub fn init_party_dir(base: &Path) -> Result<()> {
    fs::create_dir_all(base.join("agents"))
        .context("Failed to create agents directory")?;
    Ok(())
}

/// Create a new agent: write agent.json, create subdirs, create Main thread.
pub fn create_agent(base: &Path, agent: &Agent) -> Result<()> {
    let agent_dir = base.join("agents").join(&agent.slug);
    fs::create_dir_all(agent_dir.join("conversations"))?;
    fs::create_dir_all(agent_dir.join("memory"))?;
    fs::create_dir_all(agent_dir.join("instructions"))?;
    fs::create_dir_all(agent_dir.join("skills"))?;
    fs::create_dir_all(agent_dir.join("data"))?;

    save_agent(base, agent)?;

    // Create default "Main" thread
    let main_thread = Thread {
        id: Uuid::new_v4(),
        label: "Main".to_string(),
        messages: vec![],
        forked_from: None,
        created_at: Utc::now(),
    };
    save_thread(base, &agent.slug, &main_thread)?;

    Ok(())
}

/// Save agent metadata to agent.json.
pub fn save_agent(base: &Path, agent: &Agent) -> Result<()> {
    let agent_dir = base.join("agents").join(&agent.slug);
    fs::create_dir_all(&agent_dir)?;
    let json = serde_json::to_string_pretty(agent)?;
    fs::write(agent_dir.join("agent.json"), json)?;
    Ok(())
}

/// Load a single agent by slug.
pub fn load_agent(base: &Path, slug: &str) -> Result<Agent> {
    let path = base.join("agents").join(slug).join("agent.json");
    let json = fs::read_to_string(&path)
        .with_context(|| format!("Failed to read agent at {}", path.display()))?;
    let agent: Agent = serde_json::from_str(&json)?;
    Ok(agent)
}

/// List all agents (reads agent.json from each subdirectory).
pub fn list_agents(base: &Path) -> Result<Vec<Agent>> {
    let agents_dir = base.join("agents");
    if !agents_dir.exists() {
        return Ok(vec![]);
    }
    let mut agents = Vec::new();
    for entry in fs::read_dir(&agents_dir)? {
        let entry = entry?;
        if entry.file_type()?.is_dir() {
            let agent_json = entry.path().join("agent.json");
            if agent_json.exists() {
                let json = fs::read_to_string(&agent_json)?;
                let agent: Agent = serde_json::from_str(&json)?;
                agents.push(agent);
            }
        }
    }
    agents.sort_by(|a, b| a.name.cmp(&b.name));
    Ok(agents)
}

/// Delete an agent and all its data.
pub fn delete_agent(base: &Path, slug: &str) -> Result<()> {
    let agent_dir = base.join("agents").join(slug);
    if agent_dir.exists() {
        fs::remove_dir_all(&agent_dir)
            .with_context(|| format!("Failed to delete agent at {}", agent_dir.display()))?;
    }
    Ok(())
}

/// Save a thread to its own JSON file in conversations/.
pub fn save_thread(base: &Path, agent_slug: &str, thread: &Thread) -> Result<()> {
    let conv_dir = base.join("agents").join(agent_slug).join("conversations");
    fs::create_dir_all(&conv_dir)?;
    let filename = slugify(&thread.label) + ".json";
    let json = serde_json::to_string_pretty(thread)?;
    fs::write(conv_dir.join(filename), json)?;
    Ok(())
}

/// Load all threads for an agent.
pub fn load_threads(base: &Path, agent_slug: &str) -> Result<Vec<Thread>> {
    let conv_dir = base.join("agents").join(agent_slug).join("conversations");
    if !conv_dir.exists() {
        return Ok(vec![]);
    }
    let mut threads = Vec::new();
    for entry in fs::read_dir(&conv_dir)? {
        let entry = entry?;
        let path = entry.path();
        if path.extension().and_then(|e| e.to_str()) == Some("json") {
            let json = fs::read_to_string(&path)?;
            let thread: Thread = serde_json::from_str(&json)?;
            threads.push(thread);
        }
    }
    threads.sort_by(|a, b| a.created_at.cmp(&b.created_at));
    Ok(threads)
}

/// Build a FileNode tree from an agent's directory on disk.
pub fn build_file_tree(base: &Path, agent_slug: &str) -> Result<FileNode> {
    let agent_dir = base.join("agents").join(agent_slug);
    build_tree_recursive(&agent_dir, agent_slug)
}

fn build_tree_recursive(path: &Path, name: &str) -> Result<FileNode> {
    if path.is_dir() {
        let mut children = Vec::new();
        let mut entries: Vec<_> = fs::read_dir(path)?
            .filter_map(|e| e.ok())
            .collect();
        entries.sort_by_key(|e| e.file_name());
        for entry in entries {
            let child_name = entry.file_name().to_string_lossy().to_string();
            let child = build_tree_recursive(&entry.path(), &child_name)?;
            children.push(child);
        }
        Ok(FileNode {
            name: name.to_string(),
            is_dir: true,
            size: None,
            children,
            expanded: false,
        })
    } else {
        let size = fs::metadata(path).ok().map(|m| m.len());
        Ok(FileNode {
            name: name.to_string(),
```

```rust
use crate::models::*;
use crate::store;
use anyhow::Result;
use crossterm::event::{self, Event, KeyCode, KeyEvent, KeyModifiers};
use std::path::PathBuf;
use std::time::Duration;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Screen {
    AgentList,
    Chat,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Pane {
    Threads,
    Messages,
    Sidebar,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SidebarTab {
    Files,
    Projects,
    Context,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum InputTarget {
    Chat,
    AgentName,
    AgentRole,
    DeleteConfirm,
}

pub struct App {
    pub running: bool,
    pub party_dir: PathBuf,
    pub screen: Screen,
    pub agents: Vec<Agent>,
    pub selected_agent: usize,
    pub active_agent: Option<usize>,
    pub focus: Pane,
    pub sidebar_tab: SidebarTab,
    pub threads: Vec<Thread>,
    pub selected_thread: usize,
    pub selected_file: usize,
    pub selected_project: usize,
    pub scroll_offset: usize,
    pub input_mode: bool,
    pub input_target: InputTarget,
    pub input_buffer: String,
    pub file_tree: Option<FileNode>,
    pub terminal_size: (u16, u16),
    // Temporary state for agent creation (stores name between prompts)
    pub pending_agent_name: Option<String>,
}

impl App {
    pub fn new(party_dir: PathBuf) -> Result<Self> {
        store::init_party_dir(&party_dir)?;
        let agents = store::list_agents(&party_dir)?;
        Ok(Self {
            running: true,
            party_dir,
            screen: Screen::AgentList,
            agents,
            selected_agent: 0,
            active_agent: None,
            focus: Pane::Messages,
            sidebar_tab: SidebarTab::Files,
            threads: vec![],
            selected_thread: 0,
            selected_file: 0,
            selected_project: 0,
            scroll_offset: 0,
            input_mode: false,
            input_target: InputTarget::Chat,
            input_buffer: String::new(),
            file_tree: None,
            terminal_size: (80, 24),
        })
    }

    pub fn handle_event(&mut self) -> Result<()> {
        if event::poll(Duration::from_millis(50))? {
            match event::read()? {
                Event::Key(key) => self.handle_key(key)?,
                Event::Resize(w, h) => self.terminal_size = (w, h),
                _ => {}
            }
        }
        Ok(())
    }

    fn handle_key(&mut self, key: KeyEvent) -> Result<()> {
        if self.input_mode {
            return self.handle_input_key(key);
        }
        match self.screen {
            Screen::AgentList => self.handle_agent_list_key(key),
            Screen::Chat => self.handle_chat_key(key),
        }
    }

    fn handle_input_key(&mut self, key: KeyEvent) -> Result<()> {
        match key.code {
            KeyCode::Enter => {
                let input = self.input_buffer.trim().to_string();
                self.input_buffer.clear();
                self.input_mode = false;
                match self.input_target {
                    InputTarget::Chat => {
                        if !input.is_empty() {
                            self.send_message(input)?;
                        }
                    }
                    InputTarget::AgentName => {
                        if !input.is_empty() {
                            self.pending_agent_name = Some(input);
                            self.input_mode = true;
                            self.input_target = InputTarget::AgentRole;
                        }
                    }
                    InputTarget::AgentRole => {
                        if let Some(name) = self.pending_agent_name.take() {
                            if !input.is_empty() {
                                self.create_agent(name, input)?;
                            }
                        }
                    }
                    InputTarget::DeleteConfirm => {
                        if input == "y" || input == "Y" {
                            self.delete_selected_agent()?;
                        }
                    }
                }
            }
            KeyCode::Esc => {
                self.input_buffer.clear();
                self.input_mode = false;
                self.pending_agent_name = None;
            }
            KeyCode::Backspace => {
                self.input_buffer.pop();
            }
            KeyCode::Char(c) => {
                self.input_buffer.push(c);
            }
            _ => {}
        }
        Ok(())
    }

    fn handle_agent_list_key(&mut self, key: KeyEvent) -> Result<()> {
        match key.code {
            KeyCode::Char('q') => self.running = false,
            KeyCode::Char('j') | KeyCode::Down => {
                if !self.agents.is_empty() {
                    self.selected_agent = (self.selected_agent + 1).min(self.agents.len() - 1);
                }
            }
            KeyCode::Char('k') | KeyCode::Up => {
                self.selected_agent = self.selected_agent.saturating_sub(1);
            }
            KeyCode::Enter => {
                if !self.agents.is_empty() {
                    self.open_agent()?;
                }
            }
            KeyCode::Char('n') => {
                self.input_mode = true;
                self.input_target = InputTarget::AgentName;
            }
            KeyCode::Char('d') => {
                if !self.agents.is_empty() {
                    self.input_mode = true;
                    self.input_target = InputTarget::DeleteConfirm;
                }
            }
            _ => {}
        }
        Ok(())
    }

    fn handle_chat_key(&mut self, key: KeyEvent) -> Result<()> {
        match key.code {
            KeyCode::Esc => self.close_agent(),
            KeyCode::Char('i') => {
                self.input_mode = true;
                self.input_target = InputTarget::Chat;
            }
            KeyCode::Tab => {
                self.focus = match self.focus {
                    Pane::Threads => Pane::Messages,
                    Pane::Messages => Pane::Sidebar,
                    Pane::Sidebar => Pane::Threads,
                };
            }
            KeyCode::BackTab => {
                self.focus = match self.focus {
                    Pane::Threads => Pane::Sidebar,
                    Pane::Messages => Pane::Threads,
                    Pane::Sidebar => Pane::Messages,
                };
            }
            KeyCode::Char('1') => self.sidebar_tab = SidebarTab::Files,
            KeyCode::Char('2') => self.sidebar_tab = SidebarTab::Projects,
            KeyCode::Char('3') => self.sidebar_tab = SidebarTab::Context,
            KeyCode::Char('j') | KeyCode::Down => self.handle_nav_down(),
            KeyCode::Char('k') | KeyCode::Up => self.handle_nav_up(),
            KeyCode::Enter => self.handle_nav_enter()?,
            KeyCode::Char('f') => {
                if self.focus == Pane::Threads {
                    self.fork_thread()?;
                }
            }
            _ => {}
        }
        Ok(())
    }

    fn handle_nav_down(&mut self) {
        match self.focus {
            Pane::Threads => {
                if !self.threads.is_empty() {
                    self.selected_thread = (self.selected_thread + 1).min(self.threads.len() - 1);
                }
            }
            Pane::Messages => {
                self.scroll_offset = self.scroll_offset.saturating_add(1);
            }
            Pane::Sidebar => match self.sidebar_tab {
                SidebarTab::Files => {
                    let max = self.flat_file_count().saturating_sub(1);
                    self.selected_file = (self.selected_file + 1).min(max);
                }
                SidebarTab::Projects => {
                    if let Some(idx) = self.active_agent {
                        let max = self.agents[idx].projects.len().saturating_sub(1);
                        self.selected_project = (self.selected_project + 1).min(max);
                    }
                }
                SidebarTab::Context => {} // navigate snapshots (future)
            },
        }
    }

    fn handle_nav_up(&mut self) {
        match self.focus {
            Pane::Threads => {
                self.selected_thread = self.selected_thread.saturating_sub(1);
            }
            Pane::Messages => {
                self.scroll_offset = self.scroll_offset.saturating_sub(1);
            }
            Pane::Sidebar => match self.sidebar_tab {
                SidebarTab::Files => {
                    self.selected_file = self.selected_file.saturating_sub(1);
                }
                SidebarTab::Projects => {
                    self.selected_project = self.selected_project.saturating_sub(1);
                }
                SidebarTab::Context => {}
            },
        }
    }

    fn handle_nav_enter(&mut self) -> Result<()> {
        match self.focus {
            Pane::Threads => {
                // Thread is already selected by navigation; just reset scroll
                self.scroll_offset = 0;
            }
            Pane::Sidebar if self.sidebar_tab == SidebarTab::Files => {
                self.toggle_file_expand();
            }
            _ => {}
        }
        Ok(())
    }

    fn open_agent(&mut self) -> Result<()> {
        let idx = self.selected_agent;
        let slug = &self.agents[idx].slug;
        self.threads = store::load_threads(&self.party_dir, slug)?;
        self.file_tree = Some(store::build_file_tree(&self.party_dir, slug)?);
        self.active_agent = Some(idx);
        self.screen = Screen::Chat;
        self.selected_thread = 0;
        self.scroll_offset = 0;
        self.focus = Pane::Messages;
        Ok(())
    }

    fn close_agent(&mut self) {
        self.screen = Screen::AgentList;
        self.active_agent = None;
        self.threads.clear();
        self.file_tree = None;
    }

    fn send_message(&mut self, content: String) -> Result<()> {
        if self.threads.is_empty() {
            return Ok(());
        }
        let msg = Message {
            id: uuid::Uuid::new_v4(),
            role: Role::You,
            content,
            timestamp: chrono::Utc::now(),
        };
        self.threads[self.selected_thread].messages.push(msg);

        // Persist to disk
        if let Some(idx) = self.active_agent {
            let slug = &self.agents[idx].slug;
            store::save_thread(&self.party_dir, slug, &self.threads[self.selected_thread])?;
        }
        Ok(())
    }

    fn create_agent(&mut self, name: String, role: String) -> Result<()> {
        let agent = Agent {
            id: uuid::Uuid::new_v4(),
            name: name.clone(),
            slug: slugify(&name),
            role,
            status: AgentStatus::Idle,
            channels: vec![],
            projects: vec![],
            context_snapshots: vec![],
            created_at: chrono::Utc::now(),
        };
        store::create_agent(&self.party_dir, &agent)?;
        self.agents = store::list_agents(&self.party_dir)?;
        // Find the new agent and open it
        if let Some(idx) = self.agents.iter().position(|a| a.slug == agent.slug) {
            self.selected_agent = idx;
            self.open_agent()?;
        }
        Ok(())
    }

    fn delete_selected_agent(&mut self) -> Result<()> {
        if self.agents.is_empty() {
            return Ok(());
        }
        let slug = self.agents[self.selected_agent].slug.clone();
        store::delete_agent(&self.party_dir, &slug)?;
        self.agents = store::list_agents(&self.party_dir)?;
        if self.selected_agent >= self.agents.len() && !self.agents.is_empty() {
            self.selected_agent = self.agents.len() - 1;
        }
        Ok(())
    }

    fn fork_thread(&mut self) -> Result<()> {
        if self.threads.is_empty() {
            return Ok(());
        }
        let parent = &self.threads[self.selected_thread];
        let parent_id = parent.id;
        let msg_idx = parent.messages.len().saturating_sub(1);
        let fork_num = self.threads.len();
        let new_thread = Thread {
            id: uuid::Uuid::new_v4(),
            label: format!("Fork {}", fork_num),
            messages: vec![],
            forked_from: Some((parent_id, msg_idx)),
            created_at: chrono::Utc::now(),
        };
        if let Some(idx) = self.active_agent {
            let slug = &self.agents[idx].slug;
            store::save_thread(&self.party_dir, slug, &new_thread)?;
        }
        self.threads.push(new_thread);
        self.selected_thread = self.threads.len() - 1;
        Ok(())
    }

    /// Count visible (flattened) file tree nodes for navigation
    fn flat_file_count(&self) -> usize {
        match &self.file_tree {
            Some(tree) => count_visible_nodes(tree),
            None => 0,
        }
    }

    /// Toggle expand/collapse on the currently selected file tree node
    fn toggle_file_expand(&mut self) {
        if let Some(ref mut tree) = self.file_tree {
            let mut idx = 0;
            toggle_node_at(tree, self.selected_file, &mut idx);
        }
    }

    /// Get the currently selected thread's messages (convenience)
    pub fn current_messages(&self) -> &[Message] {
        if self.threads.is_empty() {
            return &[];
        }
        &self.threads[self.selected_thread].messages
    }

    /// Get the active agent (convenience)
    pub fn active_agent_ref(&self) -> Option<&Agent> {
        self.active_agent.map(|idx| &self.agents[idx])
    }

    /// Get the prompt label for input mode
    pub fn input_prompt(&self) -> &str {
        match self.input_target {
            InputTarget::Chat => "",
            InputTarget::AgentName => "Agent name: ",
            InputTarget::AgentRole => "Role: ",
            InputTarget::DeleteConfirm => "Delete? (y/n): ",
        }
    }
}

fn count_visible_nodes(node: &FileNode) -> usize {
    let mut count = 1; // this node
    if node.is_dir && node.expanded {
        for child in &node.children {
            count += count_visible_nodes(child);
        }
    }
    count
}

fn toggle_node_at(node: &mut FileNode, target: usize, current: &mut usize) -> bool {
    if *current == target {
        if node.is_dir {
            node.expanded = !node.expanded;
        }
        return true;
    }
    *current += 1;
    if node.is_dir && node.expanded {
        for child in &mut node.children {
            if toggle_node_at(child, target, current) {
                return true;
            }
        }
    }
    false
}
```

```rust
use ratatui::Frame;

pub fn render_messages(frame: &mut Frame, area: Rect, app: &App) {
    let mut lines: Vec<Line> = Vec::new();

    // Thread name header
    if !app.threads.is_empty() {
        let thread_label = &app.threads[app.selected_thread].label;
        lines.push(Line::from(vec![
            Span::styled("─── ", theme::dim()),
            Span::styled(thread_label.as_str(), theme::dim()),
            Span::styled(" ───", theme::dim()),
        ]));
        lines.push(Line::from(""));
    }

    let messages = app.current_messages();
    for msg in messages.iter() {
        let time = msg.timestamp.format("%H:%M").to_string();

        // Role + timestamp line
        lines.push(Line::from(vec![
            Span::styled("  ", theme::text()),
            Span::styled(msg.role.label(), theme::role_style(msg.role.label())),
            Span::styled("  ", theme::text()),
            Span::styled(time, theme::dim()),
        ]));

        // Content lines (indented)
        for content_line in msg.content.lines() {
            lines.push(Line::from(vec![
                Span::styled("    ", theme::text()),
                Span::styled(content_line.to_string(), theme::text()),
            ]));
        }

        // Blank line between messages
        lines.push(Line::from(""));
    }

    if messages.is_empty() {
        lines.push(Line::from(""));
        lines.push(Line::from(Span::styled(
            "No messages yet. Press 'i' to type.",
            theme::dim(),
        )));
    }

    // Apply scroll offset
    let visible_height = area.height as usize;
    let total_lines = lines.len();
    let max_scroll = total_lines.saturating_sub(visible_height);
    let scroll = app.scroll_offset.min(max_scroll);

    let paragraph = Paragraph::new(lines).scroll((scroll as u16, 0));

    // Scroll indicator
    if max_scroll > 0 {
        let indicator = format!("[{}/{}]", scroll, max_scroll);
        let ind_area = Rect {
            x: area.x + area.width.saturating_sub(indicator.len() as u16 + 1),
            y: area.y,
            width: indicator.len() as u16,
            height: 1,
        };
        let ind_widget = Paragraph::new(Span::styled(indicator, theme::dim()));
        frame.render_widget(paragraph, area);
        frame.render_widget(ind_widget, ind_area);
    } else {
        frame.render_widget(paragraph, area);
    }
}
```

```rust
pub fn render_files(frame: &mut Frame, area: Rect, app: &App) {
    let mut lines: Vec<Line> = Vec::new();

    match &app.file_tree {
        Some(tree) => {
            let mut flat_idx = 0;
            render_node(&mut lines, tree, 0, app.selected_file, &mut flat_idx);
        }
        None => {
            lines.push(Line::from(Span::styled("No file tree", theme::dim())));
        }
    }

    let paragraph = Paragraph::new(lines);
    frame.render_widget(paragraph, area);
}

fn render_node(
    lines: &mut Vec<Line<'static>>,
    node: &FileNode,
    depth: usize,
    selected: usize,
    flat_idx: &mut usize,
) {
    let is_selected = *flat_idx == selected;
    let indent = "  ".repeat(depth);
    let cursor = if is_selected { "▸ " } else { "  " };

    let icon = if node.is_dir {
        if node.expanded { "▾ " } else { "▸ " }
    } else {
        "◇ "
    };

    let name_style = if is_selected {
        theme::accent_bold()
    } else if node.is_dir {
        theme::bold()
    } else {
        theme::text()
    };

    let mut spans = vec![
        Span::styled(format!("{}{}", indent, cursor), if is_selected { theme::accent() } else { theme::text() }),
        Span::styled(icon.to_string(), theme::dim()),
        Span::styled(node.name.clone(), name_style),
    ];

    // File size
    if let Some(size) = node.size {
        let size_str = format_size(size);
        spans.push(Span::styled(format!("  {}", size_str), theme::dim()));
    }

    lines.push(Line::from(spans));
    *flat_idx += 1;

    // Render children if expanded
    if node.is_dir && node.expanded {
        for child in &node.children {
            render_node(lines, child, depth + 1, selected, flat_idx);
        }
    }
}

fn format_size(bytes: u64) -> String {
    if bytes < 1024 {
        format!("{}B", bytes)
    } else if bytes < 1024 * 1024 {
        format!("{:.1}K", bytes as f64 / 1024.0)
    } else {
        format!("{:.1}M", bytes as f64 / (1024.0 * 1024.0))
    }
}
```

```rust
use crate::models::*;
use crate::theme;
use ratatui::text::{Line, Span};

/// Parse a single line of message content for inline block syntax.
/// Returns an empty vec if the line is plain text.
pub fn parse_inline_blocks(line: &str) -> Vec<InlineBlock> {
    let trimmed = line.trim();

    if let Some(block) = try_parse_metric(trimmed) {
        return vec![block];
    }
    if let Some(block) = try_parse_table(trimmed) {
        return vec![block];
    }
    if let Some(block) = try_parse_approval(trimmed) {
        return vec![block];
    }
    if let Some(block) = try_parse_code(trimmed) {
        return vec![block];
    }

    vec![]
}

fn try_parse_metric(line: &str) -> Option<InlineBlock> {
    if !line.starts_with("[metric:") || !line.ends_with(']') {
        return None;
    }
    let inner = &line[8..line.len() - 1]; // strip "[metric:" and "]"
    let parts: Vec<&str> = inner.split('|').map(|s| s.trim()).collect();
    match parts.len() {
        2 => Some(InlineBlock::Metric {
            label: parts[0].to_string(),
            value: parts[1].to_string(),
            delta: None,
        }),
        3 => Some(InlineBlock::Metric {
            label: parts[0].to_string(),
            value: parts[1].to_string(),
            delta: Some(parts[2].to_string()),
        }),
        _ => None,
    }
}

fn try_parse_table(line: &str) -> Option<InlineBlock> {
    if !line.starts_with("[table:") || !line.ends_with(']') {
        return None;
    }
    let inner = &line[7..line.len() - 1]; // strip "[table:" and "]"
    let sections: Vec<&str> = inner.split('|').map(|s| s.trim()).collect();
    if sections.len() < 2 {
        return None;
    }
    let headers: Vec<String> = sections[0].split(',').map(|s| s.trim().to_string()).collect();
    let rows: Vec<Vec<String>> = sections[1..]
        .iter()
        .map(|row| row.split(',').map(|s| s.trim().to_string()).collect())
        .collect();
    Some(InlineBlock::Table { headers, rows })
}

fn try_parse_approval(line: &str) -> Option<InlineBlock> {
    if !line.starts_with("[approval:") || !line.ends_with(']') {
        return None;
    }
    let inner = &line[10..line.len() - 1]; // strip "[approval:" and "]"
    let parts: Vec<&str> = inner.split('|').map(|s| s.trim()).collect();
    if parts.len() != 2 {
        return None;
    }
    let status = match parts[1].to_lowercase().as_str() {
        "pending" => ApprovalStatus::Pending,
        "approved" => ApprovalStatus::Approved,
        "rejected" => ApprovalStatus::Rejected,
        _ => return None,
    };
    Some(InlineBlock::Approval {
        action: parts[0].to_string(),
        status,
    })
}

fn try_parse_code(line: &str) -> Option<InlineBlock> {
    // [code: lang] content  OR  [code] content
    if !line.starts_with("[code") {
        return None;
    }
    let after_code = &line[5..]; // after "[code"
    if let Some(rest) = after_code.strip_prefix(':') {
        // [code: lang] content
        let bracket_end = rest.find(']')?;
        let language = rest[..bracket_end].trim().to_string();
        let code = rest[bracket_end + 1..].trim().to_string();
        Some(InlineBlock::Code {
            language: if language.is_empty() { None } else { Some(language) },
            code,
        })
    } else if let Some(rest) = after_code.strip_prefix(']') {
        // [code] content
        let code = rest.trim().to_string();
        Some(InlineBlock::Code {
            language: None,
            code,
        })
    } else {
        None
    }
}

/// Render an InlineBlock as terminal Lines for display in the message pane.
pub fn render_block(block: &InlineBlock) -> Vec<Line<'static>> {
    match block {
        InlineBlock::Metric { label, value, delta } => render_metric(label, value, delta.as_deref()),
        InlineBlock::Table { headers, rows } => render_table(headers, rows),
        InlineBlock::Approval { action, status } => render_approval(action, status),
        InlineBlock::Code { language, code } => render_code(language.as_deref(), code),
    }
}

fn render_metric(label: &str, value: &str, delta: Option<&str>) -> Vec<Line<'static>> {
    let mut spans = vec![
        Span::styled("    ┌ ", theme::dim()),
        Span::styled(label.to_string(), theme::dim()),
        Span::styled("  ", theme::text()),
        Span::styled(value.to_string(), theme::bold()),
    ];
    if let Some(d) = delta {
        let delta_style = if d.starts_with('+') { theme::status_style("ACTIVE") } else { theme::alert() };
        spans.push(Span::styled("  ", theme::text()));
        spans.push(Span::styled(d.to_string(), delta_style));
    }
    vec![Line::from(spans)]
}

fn render_table(headers: &[String], rows: &[Vec<String>]) -> Vec<Line<'static>> {
    let mut lines = Vec::new();

    // Calculate column widths
    let col_count = headers.len();
    let mut widths: Vec<usize> = headers.iter().map(|h| h.len()).collect();
    for row in rows {
        for (i, cell) in row.iter().enumerate() {
            if i < widths.len() {
                widths[i] = widths[i].max(cell.len());
            }
        }
    }

    // Header line
    let mut header_spans = vec![Span::styled("    ┌ ", theme::dim())];
    for (i, h) in headers.iter().enumerate() {
        if i > 0 {
            header_spans.push(Span::styled("  ", theme::dim()));
        }
        header_spans.push(Span::styled(
            format!("{:width$}", h, width = widths[i]),
            theme::dim(),
        ));
    }
    lines.push(Line::from(header_spans));

    // Data rows
    for row in rows {
        let mut row_spans = vec![Span::styled("    │ ", theme::dim())];
        for (i, cell) in row.iter().enumerate() {
            if i > 0 {
                row_spans.push(Span::styled("  ", theme::dim()));
            }
            let w = if i < widths.len() { widths[i] } else { cell.len() };
            row_spans.push(Span::styled(
                format!("{:width$}", cell, width = w),
                theme::text(),
            ));
        }
        lines.push(Line::from(row_spans));
    }

    lines
}

fn render_approval(action: &str, status: &ApprovalStatus) -> Vec<Line<'static>> {
    let (symbol, style) = match status {
        ApprovalStatus::Pending => ("?", theme::status_style("PAUSED")),
        ApprovalStatus::Approved => ("✓", theme::status_style("ACTIVE")),
        ApprovalStatus::Rejected => ("✗", theme::alert()),
    };
    vec![Line::from(vec![
        Span::styled("    ┌ ", theme::dim()),
        Span::styled(format!("({}) ", symbol), style),
        Span::styled(action.to_string(), theme::text()),
    ])]
}

fn render_code(language: Option<&str>, code: &str) -> Vec<Line<'static>> {
    let mut lines = Vec::new();
    let lang_label = language.unwrap_or("code");
    lines.push(Line::from(vec![
        Span::styled("    ┌─ ", theme::dim()),
        Span::styled(lang_label.to_string(), theme::dim()),
    ]));
    for code_line in code.lines() {
        lines.push(Line::from(vec![
            Span::styled("    │ ", theme::dim()),
            Span::styled(code_line.to_string(), theme::text()),
        ]));
    }
    lines.push(Line::from(Span::styled("    └─", theme::dim())));
    lines
}
```

```rust
use crate::app::App;
use crate::theme;

use ratatui::Frame;

pub fn render_messages(frame: &mut Frame, area: Rect, app: &App) {
    let mut lines: Vec<Line> = Vec::new();

    // Thread name header
    if !app.threads.is_empty() {
        let thread_label = &app.threads[app.selected_thread].label;
        lines.push(Line::from(vec![
            Span::styled("─── ", theme::dim()),
            Span::styled(thread_label.as_str(), theme::dim()),
            Span::styled(" ───", theme::dim()),
        ]));
        lines.push(Line::from(""));
    }

    let messages = app.current_messages();
    for msg in messages.iter() {
        let time = msg.timestamp.format("%H:%M").to_string();

        // Role + timestamp line
        lines.push(Line::from(vec![
            Span::styled("  ", theme::text()),
            Span::styled(msg.role.label(), theme::role_style(msg.role.label())),
            Span::styled("  ", theme::text()),
            Span::styled(time, theme::dim()),
        ]));

        // Content lines — check each line for inline blocks
        for content_line in msg.content.lines() {
            let parsed = blocks::parse_inline_blocks(content_line);
            if parsed.is_empty() {
                // Plain text
                lines.push(Line::from(vec![
                    Span::styled("    ", theme::text()),
                    Span::styled(content_line.to_string(), theme::text()),
                ]));
            } else {
                // Render each block
                for block in &parsed {
                    lines.push(Line::from("")); // blank line before block
                    lines.extend(blocks::render_block(block));
                }
            }
        }

        // Blank line between messages
        lines.push(Line::from(""));
    }

    if messages.is_empty() {
        lines.push(Line::from(""));
        lines.push(Line::from(Span::styled(
            "No messages yet. Press 'i' to type.",
            theme::dim(),
        )));
    }

    // Scroll
    let visible_height = area.height as usize;
    let total_lines = lines.len();
    let max_scroll = total_lines.saturating_sub(visible_height);
    let scroll = app.scroll_offset.min(max_scroll);

    let paragraph = Paragraph::new(lines).scroll((scroll as u16, 0));

    if max_scroll > 0 {
        let indicator = format!("[{}/{}]", scroll, max_scroll);
        let ind_area = Rect {
            x: area.x + area.width.saturating_sub(indicator.len() as u16 + 1),
            y: area.y,
            width: indicator.len() as u16,
            height: 1,
        };
        let ind_widget = Paragraph::new(Span::styled(indicator, theme::dim()));
        frame.render_widget(paragraph, area);
        frame.render_widget(ind_widget, ind_area);
    } else {
        frame.render_widget(paragraph, area);
    }
}
```
